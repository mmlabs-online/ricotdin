function Button({ variant = 'secondary', className = '', ...props }: ButtonProps) {

const screenPaths: Record<Screen, string> = {
  meetings: '/meetings',
  detail: '/meetings/weekly-product-sync',

  chat: '/chat',

function getScreen(pathname: string): Screen {
  if (/^\/meetings\/[^/]+/.test(pathname)) return 'detail'
  if (pathname.startsWith('/record')) return 'record'
  if (pathname.startsWith('/chat')) return 'chat'
  if (pathname.startsWith('/profile')) return 'profile'
  if (pathname.startsWith('/admin')) return 'admin'
  if (pathname.startsWith('/sign-in')) return 'auth'
  return 'meetings'
}
